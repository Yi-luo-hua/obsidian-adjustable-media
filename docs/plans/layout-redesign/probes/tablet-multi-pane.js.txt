(async()=>{
  const wait=ms=>new Promise(r=>setTimeout(r,ms)),original=app.workspace.activeLeaf,leaf=app.workspace.getLeaf('split','vertical');
  const sourceBefore=await app.vault.read(app.vault.getAbstractFileByPath('03 文字输入.md'));
  const result={samples:[],sourceUnchanged:true};
  try{
    await original.openFile(app.vault.getAbstractFileByPath('03 文字输入.md'),{state:{mode:'preview'},active:false});
    await leaf.openFile(app.vault.getAbstractFileByPath('03 文字输入.md'),{state:{mode:'source'},active:false});app.workspace.setActiveLeaf(leaf,{focus:false});await wait(600);
    leaf.view.editor.cm.dispatch({selection:{anchor:leaf.view.editor.cm.state.doc.length}});await wait(200);
    for(const l of [original,leaf]){
      const v=l.view,container=v.getMode()==='source'?v.editor.cm.dom:v.previewMode.containerEl;
      result.samples.push({mode:v.getMode(),pane:container.getBoundingClientRect().toJSON(),display:getComputedStyle(container).display,layouts:[...container.querySelectorAll('.vml-layout')].map(el=>el.getBoundingClientRect().toJSON()),buffer:v.editor.cm.state.doc.toString()});
    }
    result.bothVisible=result.samples.every(s=>s.pane.width>0&&s.pane.height>0&&s.layouts.some(r=>r.width>0&&r.height>0));
    result.sourceUnchanged=await app.vault.read(leaf.view.file)===sourceBefore;
  }finally{leaf.detach();app.workspace.setActiveLeaf(original,{focus:false});await original.openFile(app.vault.getAbstractFileByPath('03 文字输入.md'),{state:{mode:'source'},active:true});await wait(450);const cm=original.view.editor.cm;cm.dispatch({selection:{anchor:cm.state.doc.length}});cm.scrollDOM.scrollTop=0;}
  return result;
})();
