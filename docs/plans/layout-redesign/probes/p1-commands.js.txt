(async () => {
  if (app.vault.getName() !== 'vml-test-vault') throw new Error('Test vault required');
  const fs = require('fs');
  const crypto = require('crypto');
  const path = 'Layout redesign P1 commands transient.md';
  if (app.vault.getAbstractFileByPath(path)) throw new Error('Probe note already exists');
  const hash = text => crypto.createHash('sha256').update(text).digest('hex');
  const base = app.vault.adapter.getBasePath();
  const beforeNotes = Object.fromEntries(app.vault.getMarkdownFiles().map(file => [file.path, hash(fs.readFileSync(base + '/' + file.path))]));
  const known = '<!-- vml -->\n![[known.png|240]]\n<!-- /vml -->';
  const unknown = '<!-- vml {"v":3,"id":"keep","groupId":"g"} -->\n![[keep.png|240]]\n<!-- /vml -->';
  const initial = 'Intro\n\n' + known + '\n\n' + unknown + '\n\nTail';
  const file = await app.vault.create(path, initial);
  const leaf = app.workspace.getLeaf('split', 'vertical');
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  let result;
  let restoreDispatch;
  let modal;
  try {
    await leaf.setViewState({ type: 'markdown', state: { file: path, mode: 'source' } });
    app.workspace.setActiveLeaf(leaf, { focus: false });
    const view = leaf.view;
    const editor = view.editor;
    const cm = editor.cm;
    if (!cm?.state || !cm.dispatch) throw new Error('Expected CM6 editor');
    const dispatched = [];
    const originalDispatch = cm.dispatch;
    cm.dispatch = function (...args) {
      const before = cm.state;
      const value = originalDispatch.apply(this, args);
      if (cm.state.doc !== before.doc) dispatched.push({ before: before.doc.toString(), after: cm.state.doc.toString() });
      return value;
    };
    restoreDispatch = () => { cm.dispatch = originalDispatch; };
    // Observation only: CodeMirror's stored StateField values in this supported host build.
    const snapshot = () => cm.state.values.find(value => value?.snapshot?.text === editor.getValue())?.snapshot;
    const beforeIdentity = snapshot();
    if (!beforeIdentity) throw new Error('Snapshot field not installed');
    editor.replaceRange('Prefix\n', { line: 0, ch: 0 });
    const afterIdentity = snapshot();
    editor.undo();
    const undoIdentity = snapshot();
    const identity = {
      afterInsertion: JSON.stringify(beforeIdentity.blocks.map(ref => ref.id)) === JSON.stringify(afterIdentity.blocks.map(ref => ref.id)),
      afterUndo: JSON.stringify(beforeIdentity.blocks.map(ref => ref.id)) === JSON.stringify(undoIdentity.blocks.map(ref => ref.id)),
      newUndoRevision: undoIdentity.revision > afterIdentity.revision,
    };
    dispatched.length = 0;
    const run = id => app.commands.commands['adjustable-media:' + id].editorCallback(editor, view);
    editor.setCursor({ line: 7, ch: 0 });
    run('remove-layout-comments-here');
    editor.setSelection({ line: 7, ch: 0 }, { line: 7, ch: editor.getLine(7).length });
    run('wrap-selection-in-layout');
    editor.setCursor({ line: 3, ch: 0 });
    run('merge-with-next-layout');
    const readonly = { preserved: editor.getValue() === initial, documentTransactions: dispatched.length };
    run('remove-layout-comments-here');
    const knownRemoval = { correct: editor.getValue() === initial.replace(known, '![[known.png|240]]'),
      documentTransactions: dispatched.length, unknownPreserved: editor.getValue().includes(unknown) };
    const beforeWrap = dispatched.length;
    editor.setSelection({ line: 2, ch: 0 }, { line: 2, ch: editor.getLine(2).length });
    run('wrap-selection-in-layout');
    const wrap = { correct: editor.getValue() === initial, documentTransactions: dispatched.length - beforeWrap };
    editor.undo();
    wrap.undo = editor.getValue() === initial.replace(known, '![[known.png|240]]');
    editor.undo();
    const undo = editor.getValue() === initial;
    await wait(2500);
    // Exercise the registered batch command, restricting its existing confirmation model to this fixture.
    const markdownFiles = app.vault.getMarkdownFiles;
    // Narrow discovery to this disposable fixture. Confirmation and writeBack are the real command.
    try {
      app.vault.getMarkdownFiles = () => [file];
      app.commands.executeCommandById('adjustable-media:remove-all-layout-comments');
      await wait(600);
    } finally {
      app.vault.getMarkdownFiles = markdownFiles;
    }
    modal = Array.from(app.workspace.containerEl.ownerDocument.querySelectorAll('.modal-container .modal'))
      .find(el => el.textContent.includes(path));
    if (!modal || !modal.textContent.includes(path + ' · 1')) throw new Error('Expected exactly one editable fixture block');
    const batchPreview = modal.textContent;
    const beforeBatch = dispatched.length;
    modal.querySelector('.modal-button-container button.mod-warning').click();
    await wait(100);
    const batch = { correct: editor.getValue() === initial.replace(known, '![[known.png|240]]'),
      documentTransactions: dispatched.length - beforeBatch, unknownPreserved: editor.getValue().includes(unknown) };
    editor.undo();
    batch.undo = editor.getValue() === initial;
    result = { vault: app.vault.getName(), identity, readonly, knownRemoval, wrap, undo, batchPreview: batchPreview ?? null,
      batchCountExcludesUnknown: batchPreview.includes(path + ' · 1'), batch,
      buildHash: hash(fs.readFileSync(base + '/.obsidian/plugins/adjustable-media/main.js')) };
  } finally {
    restoreDispatch?.();
    modal?.querySelector('.modal-button-container button:not(.mod-warning)')?.click();
    leaf.detach();
    await wait(600);
    await app.vault.trash(file, false);
  }
  const afterNotes = Object.fromEntries(Object.keys(beforeNotes).map(path => [path, hash(fs.readFileSync(base + '/' + path))]));
  result.existingNotesUnchanged = JSON.stringify(beforeNotes) === JSON.stringify(afterNotes);
  window.__layoutP1Commands = result;
  console.log(JSON.stringify(result));
})().catch(error => { window.__layoutP1Commands = { error: String(error), stack: error.stack }; console.error(error); });
