(async () => {
  if(app.vault.getName()!=='vml-test-vault')throw new Error('Test vault required');
  const fs=require('fs'),crypto=require('crypto'),remote=require('@electron/remote');
  const root='E:/TOOLS/obsidian-adjustable-media/',base=app.vault.adapter.getBasePath();
  const hash=data=>crypto.createHash('sha256').update(data).digest('hex');
  const originals=Object.fromEntries(app.vault.getMarkdownFiles().map(f=>[f.path,hash(fs.readFileSync(base+'/'+f.path))]));
  const prefix='<!-- vml {"v":2,"type":"text"} -->\nProbe\n<!-- /vml -->\n\n';
  const path='W07 real structural input transient.md';
  if(app.vault.getAbstractFileByPath(path))throw new Error('Probe file exists');
  const file=await app.vault.create(path,prefix),previous=app.workspace.activeLeaf,leaf=app.workspace.getLeaf(true);
  const wait=ms=>new Promise(resolve=>setTimeout(resolve,ms));
  const dbg=remote.getCurrentWebContents().debugger,attached=!dbg.isAttached();
  const result={buildHash:hash(fs.readFileSync(base+'/.obsidian/plugins/adjustable-media/main.js')),samples:[]};
  let cm,dispatch,finish,start,recording=false,positions=[];
  const position=()=>{
    const line=cm.state.doc.toString().split('\n').findIndex(l=>l==='W07 END SENTINEL');
    return cm.lineBlockAt(cm.state.doc.line(line+1).from).top;
  };
  try{
    if(attached)dbg.attach('1.3');
    await dbg.sendCommand('Emulation.setFocusEmulationEnabled',{enabled:true});
    await leaf.setViewState({type:'markdown',state:{file:path,mode:'source'}});app.workspace.setActiveLeaf(leaf,{focus:false});
    await wait(150);cm=leaf.view.editor.cm;dispatch=cm.dispatch;
    cm.dispatch=function(...specs){
      const host=specs.some(s=>(s.effects?(Array.isArray(s.effects)?s.effects:[s.effects]):[]).some(e=>e.value?.snapshot&&Array.isArray(e.value?.sections)));
      const answer=dispatch.apply(this,specs);
      if(host&&finish){const done=finish;finish=null;cm.requestMeasure({read:()=>({at:performance.now(),top:position()}),write:m=>{recording=false;done({latencyMs:m.at-start,finalTop:m.top,frames:positions})}})}
      return answer;
    };
    const frame=()=>{if(recording)positions.push({at:performance.now()-start,top:position()});if(!result.finished)requestAnimationFrame(frame)};
    requestAnimationFrame(frame);
    const scenarios=[
      {name:'quote before indented code',lines:['plain','    code one','','    code two'],line:0,from:0,to:5,text:'> quote'},
      {name:'quote before code within list',lines:['- alpha','  plain','      code one','','      code two'],line:1,from:2,to:7,text:'> quote'},
      {name:'nested list indentation',lines:['- alpha',' * beta'],line:1,from:0,to:0,text:' '},
      {name:'insert blank line',lines:['alpha','beta'],line:0,from:5,to:5,key:'Enter',code:'Enter'},
      {name:'delete blank line',lines:['alpha','','beta'],line:1,from:0,to:0,key:'Delete',code:'Delete'},
      {name:'insert fence',lines:['plain','code one','','code two'],line:0,from:0,to:5,text:'```'},
      {name:'remove fence',lines:['```','code one','','code two','```'],line:0,from:0,to:3,key:'Backspace',code:'Backspace'},
    ];
    const real=await app.vault.read(app.vault.getAbstractFileByPath('wrap-test.md'));
    const large=Array.from({length:10000},(_,i)=>i%2?'':'Filler '+i).join('\n');
    for(const [size,filler] of [['real',real],['10000-lines',large]]){
      for(const scenario of scenarios){
        for(let repetition=0;repetition<5;repetition++){
          const text=prefix+scenario.lines.join('\n')+'\n\nW07 END SENTINEL\n\n'+filler;
          cm.dispatch({changes:{from:0,to:cm.state.doc.length,insert:text},selection:{anchor:0}});await wait(size==='real'?250:450);
          const line=cm.state.doc.line(scenario.line+5);
          cm.dispatch({selection:{anchor:line.from+scenario.from,head:line.from+scenario.to}});
          cm.contentDOM.focus();await wait(30);
          const before=position();positions=[];
          const measured=new Promise((resolve,reject)=>{const timer=setTimeout(()=>reject(new Error('No host result after real input')),5000);finish=sample=>{clearTimeout(timer);resolve(sample)}});
          start=performance.now();recording=true;
          if(scenario.text!==undefined)await dbg.sendCommand('Input.insertText',{text:scenario.text});
          else{
            await dbg.sendCommand('Input.dispatchKeyEvent',{type:'keyDown',key:scenario.key,code:scenario.code,windowsVirtualKeyCode:scenario.key==='Enter'?13:scenario.key==='Delete'?46:8});
            await dbg.sendCommand('Input.dispatchKeyEvent',{type:'keyUp',key:scenario.key,code:scenario.code,windowsVirtualKeyCode:scenario.key==='Enter'?13:scenario.key==='Delete'?46:8});
          }
          const sample=await measured;
          const edited=cm.state.doc.line(scenario.line+5).text;
          result.samples.push({size,name:scenario.name,repetition,beforeTop:before,edited,...sample});
        }
      }
    }
  }finally{
    result.finished=true;recording=false;if(cm&&dispatch)cm.dispatch=dispatch;
    await dbg.sendCommand('Emulation.setFocusEmulationEnabled',{enabled:false});if(attached)dbg.detach();
    leaf.detach();await wait(500);await app.vault.process(file,()=>prefix);
    result.probeRestored=hash(fs.readFileSync(base+'/'+path))===hash(prefix);await app.vault.trash(file,true);
    if(previous)app.workspace.setActiveLeaf(previous,{focus:false});
    result.notesUnchanged=Object.entries(originals).every(([p,h])=>hash(fs.readFileSync(base+'/'+p))===h);
    fs.writeFileSync(root+'docs/plans/layout-redesign/probes/'+(window.__w07EvidencePrefix || 'w07-final')+'-structural-input.json',JSON.stringify(result,null,2));
    window.__w07Structural={samples:result.samples.length,notesUnchanged:result.notesUnchanged};
  }
})().catch(error=>{window.__w07Structural={error:String(error),stack:error.stack};console.error(error)});
