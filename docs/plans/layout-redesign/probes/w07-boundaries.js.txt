(async () => {
  if (app.vault.getName() !== 'vml-test-vault') throw new Error('Test vault required');
  const fs = require('fs'), crypto = require('crypto'), remote = require('@electron/remote');
  const root = 'E:/TOOLS/obsidian-adjustable-media/docs/plans/layout-redesign/probes/';
  const base = app.vault.adapter.getBasePath();
  const hash = data => crypto.createHash('sha256').update(data).digest('hex');
  const original = Object.fromEntries(app.vault.getMarkdownFiles().map(f => [f.path, hash(fs.readFileSync(base + '/' + f.path))]));
  const cases = [
    ['mixed bullets', ['- a', ' * b']], ['ordered indent', ['1. a', '  - b']],
    ['nested list', ['1. a', '   - b']], ['paragraph list quote', ['Text:', '- a', '> quote']],
    ['lazy quote', ['> quote', 'continued']], ['lazy list', ['- a', 'continued']],
    ['quote then code', ['> quote', '    code one', '', '', '    code two', '', 'After.']],
    ['list paragraphs', ['- a', '', '    para a', '', '    para b']],
    ['list code', ['- a', '', '      code one', '', '      code two', '', 'After.']],
    ['nested paragraphs', ['- a', '  - b', '', '      para a', '', '      para b']],
    ['heading ends list', ['- a', '## Heading', '', '    code one', '', '', '    code two']],
    ['rule ends list', ['- a', '***', '', '    code one', '', '    code two']],
    ['heading in list', ['- a', '  ## In item', '      code one', '', '      code two']],
    ['quote in list', ['- a', '  > quote', '      code one', '', '      code two']],
    ['fence ends list', ['- a', '```', 'x', '```', '    code one', '', '', '    code two']],
    ['fence in list code', ['- a', '', '  ```', '  x', '  ```', '', '      code one', '', '      code two']],
    ['fence in list paragraphs', ['- a', '', '  ```', '  x', '  ```', '', '    para one', '', '    para two']],
    ['second fence ends list', ['- a', '', '  ```', '  x', '  ```', '```', 'y', '```', '    code one', '', '', '    code two']],
    ['two item fences', ['- a', '', '  ```', '  x', '  ```', '  ```', '  y', '  ```', '    para one', '', '    para two']],
    ['math stays in list', ['- a', '', '  ```', '  x', '  ```', '$$', 'y', '$$', '    para one', '', '', '    para two']],
    ['html stays in list', ['- a', '', '  ```', '  x', '  ```', '<!--', 'c', '-->', '    para one', '', '', '    para two']],
    ['percent ends list', ['- a', '', '  ```', '  x', '  ```', '%%', 'c', '%%', '    code one', '', '', '    code two']],
    ['headings', ['## Heading', 'Text.', '# Title', 'Text.']],
    ['indented code', ['Text.', '', '    code one', '', '', '    code two', '', 'After.']],
  ];
  const path = 'W07 boundary probe transient.md';
  if (app.vault.getAbstractFileByPath(path)) throw new Error('Probe file already exists');
  const previous = app.workspace.activeLeaf;
  const file = await app.vault.create(path, 'W07 probe');
  const leaf = app.workspace.getLeaf(true);
  const result = { version: remote.app.getVersion(), cases: [], originalNotesUnchanged: false };
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  try {
    for (const [name, lines] of cases) {
      const text = lines.join('\n');
      const metadata = await app.metadataCache.computeMetadataAsync(new TextEncoder().encode(text).buffer);
      await app.vault.process(file, () => text);
      await leaf.setViewState({ type: 'markdown', state: { file: path, mode: 'preview' } });
      app.workspace.setActiveLeaf(leaf, { focus: false });
      for (let n = 0; n < 50 && leaf.view.previewMode.renderer.text !== text; n++) await wait(20);
      await wait(100);
      const renderer = leaf.view.previewMode.renderer;
      const sections = renderer.sections.map(s => ({ from: s.start, to: s.end, lines: s.lines,
        keys: Object.keys(s).filter(k => /line|type|start|end/.test(k)), dom: s.el?.outerHTML.slice(0, 1600) }));
      result.cases.push({ name, lines, sections: metadata.sections, readingTextMatches: renderer.text === text, reading: sections });
    }
  } finally {
    leaf.detach();
    if (previous) app.workspace.setActiveLeaf(previous, { focus: false });
    await app.vault.process(file, () => 'W07 probe');
    result.probeRestored = hash(fs.readFileSync(base + '/' + path)) === hash('W07 probe');
    await app.vault.trash(file, true);
    result.originalNotesUnchanged = Object.entries(original).every(([p, h]) => hash(fs.readFileSync(base + '/' + p)) === h);
    fs.writeFileSync(root + 'w07-boundaries.json', JSON.stringify(result, null, 2));
    window.__w07Boundaries = { cases: result.cases.length, originalNotesUnchanged: result.originalNotesUnchanged };
  }
})().catch(error => { window.__w07Boundaries = { error: String(error), stack: error.stack }; console.error(error); });
