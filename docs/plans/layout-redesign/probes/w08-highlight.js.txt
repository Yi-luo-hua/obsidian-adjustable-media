(async () => {
  if (app.vault.getName() !== 'vml-test-vault') throw new Error('Test vault required');
  window.w08HighlightResult = null;
  const fs = require('fs'), crypto = require('crypto'), remote = require('@electron/remote');
  const root = 'E:/TOOLS/obsidian-adjustable-media/', base = app.vault.adapter.getBasePath();
  const hash = value => crypto.createHash('sha256').update(value).digest('hex');
  const originals = Object.fromEntries(app.vault.getMarkdownFiles().map(f => [f.path, hash(fs.readFileSync(base + '/' + f.path))]));
  const dataPath = base + '/.obsidian/plugins/adjustable-media/data.json';
  const data = fs.existsSync(dataPath) ? fs.readFileSync(dataPath) : null;
  const hotkeysPath = base + '/.obsidian/hotkeys.json';
  const hotkeys = fs.existsSync(hotkeysPath) ? hash(fs.readFileSync(hotkeysPath)) : null;
  let plugin = app.plugins.plugins['adjustable-media']; const saved = { ...plugin.settings };
  let tab = app.setting.pluginTabs.find(t => t.id === 'adjustable-media');
  const path = 'W08 highlight transient.md';
  if (app.vault.getAbstractFileByPath(path)) throw new Error('Probe file exists');
  const textBlock = '<!-- vml {"v":2,"type":"text"} -->\nRepeated editable text\n<!-- /vml -->';
  const source = textBlock + '\n\nOutside first\n\n' + textBlock + '\n\nOutside second\n\n<!-- vml -->\n![[attachments/square-1x1.png]]\n<!-- /vml -->\n\nOutside media\n\n<!-- vml -->\nEditable left column\n![[attachments/square-1x1.png]]\nRight column\n<!-- /vml -->\n\nOutside columns\n';
  const file = await app.vault.create(path, source), previous = app.workspace.activeLeaf;
  const leaves = [], dbg = remote.getCurrentWebContents().debugger, attached = !dbg.isAttached();
  const result = { buildHash: hash(fs.readFileSync(base + '/.obsidian/plugins/adjustable-media/main.js')), cssHash: hash(fs.readFileSync(base + '/.obsidian/plugins/adjustable-media/styles.css')), hostVersion: remote.getCurrentWindow().getTitle().match(/Obsidian ([\d.]+)$/)?.[1], installerVersion: remote.app.getVersion(), samples: [] };
  const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
  const assert = (condition, message) => { if (!condition) throw new Error(message); };
  const click = async el => {
    let r = el.getBoundingClientRect();
    if (r.bottom < 0 || r.top > innerHeight) { el.scrollIntoView({ block: 'center' }); await wait(100); r = el.getBoundingClientRect(); }
    const clip = el.closest('.vml-wrap-proxy')?.getBoundingClientRect();
    const scroller = el.closest('.cm-scroller')?.getBoundingClientRect();
    const top = Math.max(r.top, clip?.top ?? 0, scroller?.top ?? 0), bottom = Math.min(r.bottom, innerHeight, scroller?.bottom ?? innerHeight);
    const x = r.x + Math.min(r.width / 2, 80), y = top + Math.min((bottom - top) / 2, 15);
    assert(bottom > top, 'Click target has no visible content');
    await dbg.sendCommand('Input.dispatchMouseEvent', { type: 'mousePressed', x, y, button: 'left', clickCount: 1 });
    await dbg.sendCommand('Input.dispatchMouseEvent', { type: 'mouseReleased', x, y, button: 'left', clickCount: 1 });
    await wait(180);
  };
  const sample = (name, cm) => {
    const widgets = [...cm.dom.querySelectorAll('[data-vml-block-id]')].map(el => {
      const layout = el.querySelector('.vml-layout'), style = getComputedStyle(layout), r = layout.getBoundingClientRect();
      return { id: el.dataset.vmlBlockId, active: el.hasClass('vml-cursor-in-block'), readonly: layout.hasClass('vml-layout--readonly'), border: style.borderTopColor, shadow: style.boxShadow, x: r.x, y: r.y, width: r.width, height: r.height };
    });
    const value = { name, option: plugin.settings.keepLayoutHighlight, focus: cm.dom.contains(document.activeElement), sourceActive: cm.dom.querySelectorAll('.vml-source-line--active').length, widgets, textHash: hash(cm.state.doc.toString()) };
    result.samples.push(value); return value;
  };
  const screenshot = async name => {
    await remote.getCurrentWebContents().capturePage(); await wait(40);
    fs.writeFileSync(root + 'docs/plans/layout-redesign/probes/w08-' + name + '.png', (await remote.getCurrentWebContents().capturePage()).toPNG());
  };
  const select = async (cm, anchor, head = anchor) => { cm.dispatch({ selection: { anchor, head }, userEvent: 'select' }); cm.focus(); await wait(150); };
  const line = (cm, value) => { for (let n = 1; n <= cm.state.doc.lines; n++) if (cm.state.doc.line(n).text === value) return cm.state.doc.line(n); throw new Error('Missing line ' + value); };
  const bordersHidden = value => value.widgets.every(w => w.border === 'rgba(0, 0, 0, 0)' && w.shadow === 'none');
  try {
    if (attached) dbg.attach('1.3');
    await dbg.sendCommand('Emulation.setFocusEmulationEnabled', { enabled: true });
    const leaf = app.workspace.getLeaf(true); leaves.push(leaf);
    await leaf.setViewState({ type: 'markdown', state: { file: path, mode: 'source' } }); app.workspace.setActiveLeaf(leaf, { focus: false }); await wait(450);
    const cm = leaf.view.editor.cm;
    assert(plugin.settings.keepLayoutHighlight === true, 'Missing saved setting should default on');
    const initial = sample('default enabled / restored start cursor', cm);
    await screenshot('default');
    assert(initial.widgets.length === 4 && initial.sourceActive === 0, 'Restored first cursor must not reveal source');
    await tab.setControlValue('keepLayoutHighlight', false); await wait(200);
    assert(JSON.parse(fs.readFileSync(dataPath, 'utf8')).keepLayoutHighlight === false, 'Off value not persisted');
    const off = sample('disabled / restored start cursor', cm); assert(bordersHidden(off), 'Restored cursor retained highlight');
    await screenshot('hidden');
    for (const before of initial.widgets) {
      const after = off.widgets.find(w => w.id === before.id);
      assert(after && ['x', 'y', 'width', 'height'].every(k => Math.abs(before[k] - after[k]) <= 0.01), 'Toggle changed layout geometry');
    }
    const first = cm.dom.querySelector('[data-vml-block-id] .vml-layout');
    const rect = first.getBoundingClientRect();
    await dbg.sendCommand('Input.dispatchMouseEvent', { type: 'mouseMoved', x: rect.x + 20, y: rect.y + 10 }); await wait(160);
    assert(bordersHidden(sample('disabled / real hover', cm)), 'Hover enabled normal highlighting');
    const handle = first.querySelector('.vml-frame__handle'); assert(handle && getComputedStyle(handle).opacity !== '0', 'Hover resize handle unavailable');
    first.addClass('vml-layout--drop-target');
    await wait(160);
    assert(getComputedStyle(first).borderTopColor !== 'rgba(0, 0, 0, 0)' && getComputedStyle(first).boxShadow !== 'none', 'Hidden normal frame suppressed drop feedback');
    first.removeClass('vml-layout--drop-target');
    await select(cm, 0); assert(sample('user caret at opening comment', cm).sourceActive === 3, 'Opening comment caret not highlighted');
    await select(cm, line(cm, 'Outside first').from); assert(sample('caret left block', cm).sourceActive === 0, 'Highlight retained after caret left');
    const outside = line(cm, 'Outside second').from;
    await select(cm, 0, outside); assert(sample('selection covers both copies / head outside', cm).sourceActive === 0, 'Selection coverage activated block');
    await select(cm, outside, 0); assert(sample('reversed selection / head in first copy', cm).sourceActive === 3, 'Head in block did not activate it');
    const Selection = cm.state.selection.constructor;
    cm.dispatch({ selection: Selection.create([Selection.cursor(0), Selection.cursor(line(cm, 'Outside first').to + 2)]), userEvent: 'select' }); await wait(180);
    const multi = sample('multiple carets', cm); assert(multi.sourceActive === 6, 'Multiple heads did not activate separate copies');
    await select(cm, line(cm, 'Outside media').from);
    const button = [...cm.dom.querySelectorAll('.vml-edit-source')].find(el => el.closest('[data-vml-block-id]').querySelector('img'));
    await click(button); assert(sample('real source button click', cm).sourceActive === 3, 'Source button did not highlight actual source');
    await screenshot('source');
    await dbg.sendCommand('Input.dispatchKeyEvent', { type: 'keyDown', key: 'ArrowDown', code: 'ArrowDown', windowsVirtualKeyCode: 40 });
    await dbg.sendCommand('Input.dispatchKeyEvent', { type: 'keyUp', key: 'ArrowDown', code: 'ArrowDown', windowsVirtualKeyCode: 40 }); await wait(160);
    assert(sample('real keyboard caret inside source', cm).sourceActive === 3, 'Keyboard caret lost highlight');
    await select(cm, line(cm, 'Outside columns').from);
    const column = [...cm.dom.querySelectorAll('.vml-layout__text--left')].find(el => el.textContent.includes('Editable left'));
    await click(column);
    const columnEditor = cm.dom.querySelector('.vml-text-editor .cm-content'); assert(columnEditor, 'Real column click did not open editor');
    const composing = sample('column focused / main head outside', cm); assert(composing.widgets.filter(w => w.active).length === 1, 'Column did not own highlight');
    await screenshot('column');
    const columnId = composing.widgets.find(w => w.active).id;
    await dbg.sendCommand('Input.imeSetComposition', { text: '测试', selectionStart: 2, selectionEnd: 2 }); await wait(100);
    assert(sample('Chinese composition', cm).widgets.some(w => w.id === columnId && w.active), 'Composition lost highlight');
    await tab.setControlValue('keepLayoutHighlight', true); await wait(100);
    assert(cm.dom.querySelector('.vml-text-editor .cm-content') === columnEditor, 'Toggle recreated column editor');
    await tab.setControlValue('keepLayoutHighlight', false); await wait(100);
    assert(cm.dom.querySelector('.vml-text-editor .cm-content') === columnEditor, 'Toggle off interrupted composition');
    await dbg.sendCommand('Input.insertText', { text: '测试' }); await wait(180);
    assert(sample('composition committed', cm).widgets.some(w => w.id === columnId && w.active), 'Committed text lost highlight identity');
    await select(cm, line(cm, 'Outside columns').from);
    await click(cm.dom.querySelector('[data-vml-block-id="' + columnId + '"] .vml-edit-source'));
    const retained = sample('source expanded / retained readonly preview', cm);
    assert(retained.widgets.some(w => w.active && w.readonly) && retained.sourceActive === 5, 'Retained preview lost source caret highlight');
    const second = app.workspace.getLeaf('split'); leaves.push(second);
    await second.setViewState({ type: 'markdown', state: { file: path, mode: 'source' } }); app.workspace.setActiveLeaf(second, { focus: false }); await wait(300);
    const cm2 = second.view.editor.cm; await select(cm2, line(cm2, 'Outside first').from);
    assert(sample('other pane / old column focus cleared', cm).widgets.every(w => !w.active), 'Inactive pane retained column highlight');
    await select(cm2, 0); assert(sample('second pane caret', cm2).sourceActive === 3, 'Second pane failed to highlight');
    assert(sample('first pane remains inactive', cm).sourceActive === 0, 'Caret state leaked across panes');
    app.workspace.setActiveLeaf(leaf, { focus: false }); await select(cm, line(cm, 'Outside columns').from);
    assert(sample('column edit finished', cm).widgets.every(w => !w.active), 'Ended edit retained highlight');
    // A long float's anchor leaves the rendered viewport; the stand-in must carry the same identity.
    const floatText = '<!-- vml {"v":2,"type":"text","wrap":"right","width":0.4} -->\n' + Array.from({ length: 110 }, (_, i) => 'Float text ' + i + '  \n').join('') + '<!-- /vml -->\n' + Array.from({ length: 180 }, (_, i) => 'Body ' + i + '\n\n').join('');
    cm.scrollDOM.scrollTop = 0;
    cm.dispatch({ changes: { from: 0, to: cm.state.doc.length, insert: floatText }, selection: { anchor: floatText.length }, userEvent: 'input' });
    cm.scrollDOM.scrollTop = 0; await wait(650);
    sample('long float at source anchor', cm);
    const float = cm.dom.querySelector('[data-vml-block-id]'); assert(float, 'Long float not rendered at anchor');
    const floatId = float.dataset.vmlBlockId;
    cm.scrollDOM.scrollTop = 1500; await wait(400);
    const proxy = cm.dom.querySelector('.vml-wrap-proxy__live'); assert(proxy && proxy.dataset.vmlBlockId === floatId, 'Stand-in lost runtime block identity');
    assert(bordersHidden(sample('stand-in / caret outside', cm)), 'Stand-in highlighted without caret');
    await screenshot('proxy-hidden');
    await click(proxy.querySelector('.vml-layout__text'));
    const proxyEditing = sample('stand-in / real column click', cm);
    assert(proxyEditing.widgets.some(w => w.id === floatId && w.active), 'Stand-in column did not activate same block');
    await screenshot('proxy-active');
    await select(cm, line(cm, 'Body 50').from); cm.scrollDOM.scrollTop = 0; await wait(350);
    assert(sample('stand-in to real widget / focus left', cm).widgets.every(w => !w.active), 'Stand-in handoff retained highlight');
    await leaf.setViewState({ type: 'markdown', state: { file: 'text-test.md', mode: 'source' } }); await wait(200);
    const switched = sample('switch to another note / stale focus cleared', leaf.view.editor.cm);
    assert(switched.widgets.every(w => !w.active) && switched.sourceActive === 0, 'Highlight leaked into another note');
    await app.plugins.disablePlugin('adjustable-media'); await app.plugins.enablePlugin('adjustable-media'); await wait(300);
    plugin = app.plugins.plugins['adjustable-media']; tab = app.setting.pluginTabs.find(t => t.id === 'adjustable-media');
    assert(plugin.settings.keepLayoutHighlight === false, 'Reload lost disabled setting');
    assert(leaf.view.editor.cm.dom.hasClass('vml-cursor-highlights'), 'Reload did not apply persisted option');
    sample('plugin reload / disabled setting retained', leaf.view.editor.cm);
    await tab.setControlValue('keepLayoutHighlight', true); await wait(150);
    assert(JSON.parse(fs.readFileSync(dataPath, 'utf8')).keepLayoutHighlight === true, 'On value not persisted');
    result.passed = true;
  } catch (error) { result.error = String(error.stack ?? error); }
  finally {
    await dbg.sendCommand('Emulation.setFocusEmulationEnabled', { enabled: false }); if (attached) dbg.detach();
    for (const leaf of leaves.reverse()) leaf.detach(); if (previous) app.workspace.setActiveLeaf(previous, { focus: false });
    plugin.settings = saved; await plugin.saveSettings();
    if (data) fs.writeFileSync(dataPath, data); else if (fs.existsSync(dataPath)) fs.unlinkSync(dataPath);
    await app.vault.trash(file, true);
    result.originalNotesUnchanged = Object.entries(originals).every(([p, h]) => fs.existsSync(base + '/' + p) && hash(fs.readFileSync(base + '/' + p)) === h);
    result.hotkeysUnchanged = hotkeys === (fs.existsSync(hotkeysPath) ? hash(fs.readFileSync(hotkeysPath)) : null);
    result.settingsRestored = data === null ? !fs.existsSync(dataPath) : fs.readFileSync(dataPath).equals(data);
    fs.writeFileSync(root + 'docs/plans/layout-redesign/probes/w08-highlight.json', JSON.stringify(result, null, 2));
    window.w08HighlightResult = { passed: result.passed, error: result.error, samples: result.samples.length, originalNotesUnchanged: result.originalNotesUnchanged };
  }
})();
